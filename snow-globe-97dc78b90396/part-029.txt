function normalizeImageDataURL(src){
 const match=/^data:([^;,]*);base64,([A-Za-z0-9+/]+={0,2})$/i.exec(src||'');
 if(!match)throw new Error('This image could not be stored as a portable picture.');
 let type=match[1].toLowerCase();if(type==='image/jpg')type='image/jpeg';if(type==='image/x-png')type='image/png';
 if(!['image/png','image/jpeg','image/webp'].includes(type)){
  if(type&&!['application/octet-stream','binary/octet-stream'].includes(type))throw new Error('Choose an actual PNG, JPEG or WebP image.');
  const signature=atob(match[2].slice(0,32));
  if(signature.startsWith('\x89PNG\r\n\x1a\n'))type='image/png';
  else if(signature.startsWith('\xff\xd8\xff'))type='image/jpeg';
  else if(signature.startsWith('RIFF')&&signature.slice(8,12)==='WEBP')type='image/webp';
  else throw new Error('This picture has no recognizable PNG, JPEG or WebP format.');
 }
 return 'data:'+type+';base64,'+match[2];
}
